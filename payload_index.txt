        document.addEventListener("DOMContentLoaded", () => {

            // SCROLL REVEAL ANIMATION
            const observerOptions = {
                threshold: 0.12, 
                rootMargin: "0px 0px -40px 0px"
            };

            const scrollObserver = new IntersectionObserver((entries) => {
                entries.forEach((entry) => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add("active");
                    }
                });
            }, observerOptions);

            const revealElements = document.querySelectorAll(".reveal-up, .reveal-left, .reveal-right");
            revealElements.forEach((el) => scrollObserver.observe(el));

            // WEB AUDIO API
            const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            function playCyberBeep(freq = 600, duration = 0.08) {
                if (audioCtx.state === 'suspended') audioCtx.resume();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.05, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + duration);
            }

            // EFEK KETIK
            const typingElement = document.getElementById("typing-text");
            if (typingElement) { 
                const roles = ["Ayoo mutualann🗿", "Tiktok/Ig gua ada di atas😁", "#Beginner Developer"];
                let roleIndex = 0, charIndex = 0, isDeleting = false;

                function typeEffect() {
                    const currentRole = roles[roleIndex];
                    if (isDeleting) {
                        typingElement.textContent = currentRole.substring(0, charIndex--);
                    } else {
                        typingElement.textContent = currentRole.substring(0, charIndex++);
                    }

                    let speed = isDeleting ? 50 : 100;

                    if (!isDeleting && charIndex === currentRole.length + 1) {
                        speed = 2000;
                        isDeleting = true;
                    } else if (isDeleting && charIndex === 0) {
                        isDeleting = false;
                        roleIndex = (roleIndex + 1) % roles.length;
                        speed = 500;
                    }
                    setTimeout(typeEffect, speed);
                }
                typeEffect();
            }

            // LOGIKA 3D ORBIT SPHERE
            const arena = document.getElementById("arena");
            const bubbleEls = document.querySelectorAll(".bubble");

            let targetX = 0.012; 
            let targetY = 0.015; 
            let rotationVelocityX = 0;
            let rotationVelocityY = 0;
            let angleX = 0; 
            let angleY = 0; 
            const radius = 180; 
            let bubbles = [];

            const phi = Math.PI * (3 - Math.sqrt(5)); 
            bubbleEls.forEach((el, i) => {
                const y = 1 - (i / (bubbleEls.length - 1)) * 2; 
                const radiusAtY = Math.sqrt(1 - y * y);
                const theta = phi * i;

                bubbles.push({
                    el: el,
                    baseX: Math.cos(theta) * radiusAtY * radius,
                    baseY: y * radius,
                    baseZ: Math.sin(theta) * radiusAtY * radius
                });
                
                el.addEventListener("click", () => playCyberBeep(600 + (i * 20)));
            });

            arena.addEventListener("mousemove", (e) => {
                const rect = arena.getBoundingClientRect();
                const x = e.clientX - rect.left - (rect.width / 2);
                const y = e.clientY - rect.top - (rect.height / 2);
                targetY = x * 0.00002; 
                targetX = -y * 0.00002;
            });
            arena.addEventListener("mouseleave", () => { 
                targetX = 0.0012; 
                targetY = 0.0015; 
            });

            function animate3D() {
                rotationVelocityX += (targetX - rotationVelocityX) * 0.05;
                rotationVelocityY += (targetY - rotationVelocityY) * 0.05;

                angleX += rotationVelocityX;
                angleY += rotationVelocityY;

                let cosX = Math.cos(angleX), sinX = Math.sin(angleX);
                let cosY = Math.cos(angleY), sinY = Math.sin(angleY);

                bubbles.forEach(b => {
                    let y1 = b.baseY * cosX - b.baseZ * sinX;
                    let z1 = b.baseY * sinX + b.baseZ * cosX;
                    let x2 = b.baseX * cosY + z1 * sinY;
                    let z2 = -b.baseX * sinY + z1 * cosY;

                    const scale = (z2 + 250) / 300; 
                    let opacity = (z2 + 100);
                    if (opacity < 0) opacity = 0; 
                    if (opacity > 1) opacity = 1; 

                    b.el.style.transform = `translate3d(${x2}px, ${y1}px, ${z2}px) scale(${scale})`;
                    b.el.style.opacity = opacity; 
                    b.el.style.zIndex = Math.round(z2 + radius);
                });
                requestAnimationFrame(animate3D);
            }
            animate3D();
        });

        async function fetchRealNews() {
            const newsContainer = document.getElementById("news-container");
            const rssApiUrl = "https://api.rss2json.com/v1/api.json?rss_url=https://www.antaranews.com/rss/politik.xml";

            try {
                const response = await fetch(rssApiUrl);
                const data = await response.json();

                if (data.status === "ok" && data.items.length > 0) {
                    newsContainer.innerHTML = ""; 

                    const articles = data.items.slice(0, 3);

                    articles.forEach((item, index) => {
                        let imageUrl = item.thumbnail || (item.enclosure && item.enclosure.link);
                        
                        if (!imageUrl || imageUrl === "") {
                            imageUrl = "https://images.unsplash.com/photo-1504711434969-e33886168f5c?auto=format&fit=crop&w=600&h=400&q=80";
                        }

                        const pubDate = new Date(item.pubDate).toLocaleDateString("id-ID", {
                            day: "numeric", month: "long", year: "numeric"
                        });

                        const cleanDescription = item.description.replace(/<[^>]*>?/gm, '').substring(0, 120) + "...";

                        const cardHTML = `
                            <div class="news-card reveal-up delay-${index + 1} active">
                                <div class="news-img">
                                    <img src="${imageUrl}" alt="${item.title}" onerror="this.src='https://images.unsplash.com/photo-1504711434969-e33886168f5c?auto=format&fit=crop&w=600&h=400&q=80'">
                                </div>
                                <div class="news-content">
                                    <span class="news-date"><i class="fa-regular fa-calendar"></i> ${pubDate}</span>
                                    <h3>${item.title}</h3>
                                    <p>${cleanDescription}</p>
                                    <a href="${item.link}" target="_blank" rel="noopener noreferrer" class="news-link">
                                        Baca di Sumber Resmi <i class="fa-solid fa-arrow-up-right-from-square"></i>
                                    </a>
                                </div>
                            </div>
                        `;
                        newsContainer.innerHTML += cardHTML;
                    });
                } else {
                    newsContainer.innerHTML = "<p style='grid-column: 1/-1; text-align: center;'>Gagal memuat berita terkini.</p>";
                }
            } catch (error) {
                console.error("Error fetching news:", error);
                newsContainer.innerHTML = "<p style='grid-column: 1/-1; text-align: center;'>Gagal terhubung ke server berita.</p>";
            }
        }

        document.addEventListener("DOMContentLoaded", () => {
            fetchRealNews();
        });

        function bukaPayload(event, url) {
            event.preventDefault(); // Mencegah layar lompat ke atas karena href="#"
            
            // Opsional: Bikin layar transisi fade-out perlahan
            document.body.style.transition = "opacity 0.3s ease";
            document.body.style.opacity = "0";

            setTimeout(() => {
                fetch(url)
                    .then(response => {
                        if (!response.ok) throw new Error('File tidak ditemukan');
                        return response.text();
                    })
                    .then(html => {
                        document.open();
                        document.write(html);
                        document.close();
                    })
                    .catch(err => {
                        console.error(err);
                        alert("Gagal memuat halaman proyek!");
                        document.body.style.opacity = "1"; // Balikin layar kalau gagal
                    });
            }, 300);
        }
