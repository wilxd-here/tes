const viaXml=async s=>{
  const targets=[
    'https://api.allorigins.win/raw?url='+encodeURIComponent(s.u),
    'https://corsproxy.io/?url='+encodeURIComponent(s.u)
  ];
  for(const url of targets){
    try{
      const r=await fetch(url,{signal:tmo(12000),cache:'no-store'});
      if(!r.ok)continue;
      const xml=await r.text();
      if(xml && /<item[\\s>]/i.test(xml))return parseXml(xml,s);
    }catch(e){}
  }
